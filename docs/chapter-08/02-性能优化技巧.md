### 3.2 请求缓存

```typescript
const cache = new Map()

async function fetchWithCache(key: string, fn: () => Promise<any>) {
  if (cache.has(key)) {
    return cache.get(key)
  }
  const result = await fn()
  cache.set(key, result)
  return result
}
```
