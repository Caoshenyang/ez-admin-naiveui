## 1. 默认布局

**src/layouts/DefaultLayout.vue**:

```vue
<script setup lang="ts">
import { ref } from 'vue'
import { useAppStore } from '@/stores/app'

const appStore = useAppStore()

const sidebarCollapsed = ref(false)

const toggleSidebar = () => {
  sidebarCollapsed.value = !sidebarCollapsed.value
}
</script>

<template>
  <div class="default-layout">
    <!-- 侧边栏 -->
    <aside
      class="sidebar"
      :class="{ collapsed: sidebarCollapsed }"
    >
      <AppLogo />
      <AppMenu :collapsed="sidebarCollapsed" />
    </aside>

    <!-- 主内容区 -->
    <div class="main-container">
      <!-- 顶部栏 -->
      <header class="header">
        <div class="header-left">
          <n-button text @click="toggleSidebar">
            <template #icon>
              <n-icon><MenuIcon /></n-icon>
            </template>
          </n-button>
          <Breadcrumb />
        </div>

        <div class="header-right">
          <Search />
          <Notification />
          <Fullscreen />
          <ThemeToggle />
          <UserDropdown />
        </div>
      </header>

      <!-- 内容区 -->
      <main class="content">
        <router-view v-slot="{ Component }">
          <transition name="fade" mode="out-in">
            <component :is="Component" :key="$route.path" />
          </transition>
        </router-view>
      </main>
    </div>
  </div>
</template>

<style scoped>
.default-layout {
  @apply flex h-screen;
}

.sidebar {
  @apply w-64 bg-dark-800 text-white transition-all;
  &.collapsed {
    @apply w-20;
  }
}

.main-container {
  @apply flex-1 flex flex-col overflow-hidden;
}

.header {
  @apply flex items-center justify-between px-6 py-4 bg-white border-b;
}

.content {
  @apply flex-1 overflow-auto p-6 bg-gray-50;
}

.fade-enter-active,
.fade-leave-active {
  transition: opacity 0.3s;
}

.fade-enter-from,
.fade-leave-to {
  opacity: 0;
}
</style>
```

### 2.2 AppMenu

**src/layouts/components/AppMenu.vue**:

```vue
<script setup lang="ts">
import { computed } from 'vue'
import { useRouter } from 'vue-router'

const router = useRouter()

const menuOptions = computed(() => {
  // 根据路由生成菜单
  return [
    {
      label: '工作台',
      key: '/dashboard',
      icon: () => h(DashboardIcon),
    },
    {
      label: '系统管理',
      key: 'system',
      icon: () => h(SettingIcon),
      children: [
        { label: '用户管理', key: '/system/user' },
        { label: '角色管理', key: '/system/role' },
      ],
    },
  ]
})

const handleMenuSelect = (key: string) => {
  router.push(key)
}
</script>

<template>
  <n-menu
    :options="menuOptions"
    :collapsed="collapsed"
    :collapsed-width="64"
    :collapsed-icon-size="22"
    @update:value="handleMenuSelect"
  />
</template>
```
