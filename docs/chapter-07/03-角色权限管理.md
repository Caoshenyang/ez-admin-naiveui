## 1. 角色管理页面

**src/views/system/RoleManagement.vue**:

```vue
<script setup lang="ts">
import { ref } from 'vue'

const columns = [
  { key: 'id', title: 'ID' },
  { key: 'name', title: '角色名称' },
  { key: 'code', title: '角色编码' },
  { key: 'description', title: '描述' },
]

const formSchema = {
  items: [
    { prop: 'name', label: '角色名称', type: 'input' },
    { prop: 'code', label: '角色编码', type: 'input' },
    { prop: 'description', label: '描述', type: 'textarea' },
  ],
}

const api = {
  list: (params) => request.get('/roles', { params }),
  create: (data) => request.post('/roles', data),
  update: (id, data) => request.put(`/roles/${id}`, data),
  delete: (id) => request.delete(`/roles/${id}`),
}
</script>

<template>
  <EzCRUD :columns="columns" :form-schema="formSchema" :api="api" />
</template>
```

## 2. 权限树组件

**src/components/PermissionTree.vue**:

```vue
<script setup lang="ts">
import { ref, computed } from 'vue'
import { NTree } from 'naive-ui'

interface Props {
  modelValue: string[]
  permissions: any[]
}

const props = defineProps<Props>()
const emit = defineEmits<{
  'update:modelValue': [value: string[]]
}>()

const checkedKeys = computed({
  get: () => props.modelValue,
  set: (value) => emit('update:modelValue', value),
})

const treeData = computed(() => {
  // 将权限列表转换为树形结构
  return transformToTree(props.permissions)
})
</script>

<template>
  <n-tree
    v-model:checked-keys="checkedKeys"
    :data="treeData"
    checkable
    cascade
  />
</template>
```
