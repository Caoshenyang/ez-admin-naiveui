## 1. 权限指令

**src/directives/permission.ts**:

```typescript
import type { Directive, DirectiveBinding } from 'vue'
import { useUserStore } from '@/stores/user'

/**
 * 权限指令
 *
 * @example
 * ```vue
 * <button v-permission="['admin']">管理员可见</button>
 * <button v-permission="['user', 'admin']">多角色</button>
 * ```
 */
const permission: Directive = {
  mounted(el: HTMLElement, binding: DirectiveBinding<string[]>) {
    const { value } = binding

    if (value && value instanceof Array && value.length > 0) {
      const userStore = useUserStore()
      const roles = userStore.roles

      const hasPermission = value.some((role) => roles.includes(role))

      if (!hasPermission) {
        // 无权限，移除元素
        el.parentNode?.removeChild(el)
      }
    } else {
      throw new Error('need roles! Like v-permission="[\'admin\']"')
    }
  },
}

export default permission
```

## 2. 权限函数

**src/composables/usePermission.ts**:

```typescript
import { useUserStore } from '@/stores/user'

/**
 * 检查是否有权限
 */
export function hasPermission(value: string[]): boolean {
  if (value && value instanceof Array && value.length > 0) {
    const userStore = useUserStore()
    const roles = userStore.roles
    return value.some((role) => roles.includes(role))
  }
  return false
}

/**
 * 检查是否为管理员
 */
export function isAdmin(): boolean {
  const userStore = useUserStore()
  return userStore.roles.includes('admin')
}

/**
 * 使用权限
 */
export function usePermission() {
  return {
    hasPermission,
    isAdmin,
  }
}
```

## 3. 使用示例

### 3.1 指令使用

```vue
<template>
  <!-- 单一权限 -->
  <button v-permission="['admin']">管理员按钮</button>

  <!-- 多个权限（满足其一即可） -->
  <button v-permission="['admin', 'editor']">编辑按钮</button>

  <!-- 数组变量 -->
  <button v-permission="allowedRoles">权限按钮</button>
</template>

<script setup lang="ts">
const allowedRoles = ['admin', 'editor']
</script>
```

### 3.2 函数使用

```vue
<script setup lang="ts">
import { hasPermission } from '@/composables/usePermission'

const canEdit = hasPermission(['admin', 'editor'])

const handleEdit = () => {
  if (!hasPermission(['admin'])) {
    message.error('没有权限')
    return
  }
  // 执行编辑操作
}
</script>

<template>
  <button v-if="canEdit">编辑</button>
</template>
```
